###############################################################################
# 10) Metrics extraction
###############################################################################

@'
import json
from pathlib import Path
for p in Path("train/logs").glob("*.log"):
    last_step = None
    final = None
    for line in p.read_text(encoding="utf-8", errors="ignore").splitlines():
        if not line.startswith("{"):
            continue
        try:
            obj = json.loads(line)
        except Exception:
            continue
        if "step" in obj:
            last_step = obj
        if obj.get("status") == "done":
            final = obj
    print(f"\n=== {p} ===")
    if last_step:
        for k in ["step", "loss", "main_loss", "aux_loss", "grad_norm", "tokens_per_sec"]:
            if k in last_step:
                print(f"{k}: {last_step[k]}")
        imag = last_step.get("imagination_stats", {})
        for k in ["avg_delta_norm", "final_latent_norm", "latent_rms", "convergence_ratio"]:
            if k in imag:
                print(f"imagination_stats.{k}: {imag[k]}")
    else:
        print("NO_STEP_PAYLOAD")
    if final:
        print(f"final: {final}")
'@ | .\pca\Scripts\python.exe - | Tee-Object -FilePath .\tmp\metrics_summary.txt
